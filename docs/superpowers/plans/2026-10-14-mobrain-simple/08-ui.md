```ts
import { Hono } from 'hono';
import { normalize, join } from 'node:path';

export function staticRoutes(uiDir: string): Hono {
  const r = new Hono();
  r.all('/api/*', (c) => c.json({ error: 'not_found' }, 404));
  r.get('*', async (c) => {
    const rel = normalize(decodeURIComponent(c.req.path)).replace(/^(\.\.(\/|$))+/, '');
    if (rel !== '/' && !rel.includes('..')) {
      const f = Bun.file(join(uiDir, rel));
      if (await f.exists()) {
        const headers: Record<string, string> = {};
        if (rel.startsWith('/assets/')) headers['cache-control'] = 'public, max-age=31536000, immutable';
        return new Response(f, { headers });
      }
    }
    return new Response(Bun.file(join(uiDir, 'index.html')), { headers: { 'content-type': 'text/html; charset=utf-8', 'cache-control': 'no-cache' } });
  });
  return r;
}
```

```ts
export type Role = 'owner' | 'admin' | 'member' | 'viewer';
export type Mode = 'personal' | 'company';

/** [[slug]], [[slug|label]], [[slug#anchor]] -> Markdown links into the same source. Code spans are left alone. */
export function wikiToLinks(md: string, source: string): string {
  return md.split(/(`[^`]*`)/).map((part) => part.startsWith('`') ? part
    : part.replace(/\[\[([^\]|#]+)(?:#[^\]|]*)?(?:\|([^\]]+))?\]\]/g, (_, slug: string, label?: string) => {
        const s = slug.trim();
        return `[${(label ?? s).trim()}](/pages/${source}/${s})`;
      })).join('');
}

export type TreeNode = { name: string; slug?: string; children?: TreeNode[] };

export function buildTree(slugs: string[]): TreeNode[] {
  const root: TreeNode = { name: '', children: [] };
  for (const slug of slugs) {
    const parts = slug.split('/');
    let node = root;
    parts.forEach((name, i) => {
      const leaf = i === parts.length - 1;
      node.children ??= [];
      let next = node.children.find((c) => c.name === name && (leaf ? c.slug !== undefined : c.children !== undefined));
      if (!next) { next = leaf ? { name, slug } : { name, children: [] }; node.children.push(next); }
      node = next;
    });
  }
  const sort = (ns: TreeNode[]): TreeNode[] => ns
    .sort((a, b) => (a.children ? 0 : 1) - (b.children ? 0 : 1) || a.name.localeCompare(b.name))
    .map((n) => (n.children ? { ...n, children: sort(n.children) } : n));
  return sort(root.children ?? []);
}

export type NavEntry = { path: string; label: string };

export function navFor(role: Role, mode: Mode): NavEntry[] {
  const out: NavEntry[] = [
    { path: '/pages', label: 'Pages' },
    { path: '/sources', label: 'Sources' },
    { path: '/schema', label: 'Schema' },
  ];
  if (role === 'owner' || role === 'admin') out.push({ path: '/activity', label: 'Activity' }, { path: '/health', label: 'Health' });
  // Owners see People & access in both modes (it holds the mode switch); admins only manage people in company mode.
  if (role === 'owner' || (role === 'admin' && mode === 'company')) out.push({ path: '/people', label: 'People & access' });
  out.push({ path: '/connect', label: 'Connect your agent' });
  return out;
}

export function ago(iso: string | null | undefined, now = new Date()): string {
  if (!iso) return 'never';
  const ms = now.getTime() - new Date(iso).getTime();
  const abs = Math.abs(ms); const future = ms < 0;
  const fmt = (n: number, u: string) => (future ? `in ${n}${u}` : `${n}${u} ago`);
  if (abs < 60_000) return future ? 'soon' : 'just now';
  if (abs < 3600_000) return fmt(Math.round(abs / 60_000), 'm');
  if (abs < 86400_000) return fmt(Math.round(abs / 3600_000), 'h');
  return fmt(Math.round(abs / 86400_000), 'd');
}
```

```ts
export class ApiError extends Error {
  constructor(readonly status: number, readonly code: string) { super(code); }
}

export async function api<T>(path: string, opts: { method?: string; json?: unknown } = {}): Promise<T> {
  const res = await fetch(path, {
    method: opts.method ?? (opts.json === undefined ? 'GET' : 'POST'),
    headers: opts.json === undefined ? {} : { 'content-type': 'application/json' },
    body: opts.json === undefined ? undefined : JSON.stringify(opts.json),
    credentials: 'same-origin',
  });
  const body = await res.json().catch(() => ({}));
  if (!res.ok) throw new ApiError(res.status, (body as { error?: string }).error ?? `http_${res.status}`);
  return body as T;
}
```

```ts
import { useEffect, useState, type MouseEvent } from 'react';

export function navigate(to: string): void {
  if (to === location.pathname + location.search) return;
  history.pushState(null, '', to);
  dispatchEvent(new PopStateEvent('popstate'));
}

export function usePath(): string {
  const [path, setPath] = useState(location.pathname);
  useEffect(() => {
    const on = () => setPath(location.pathname);
    addEventListener('popstate', on);
    return () => removeEventListener('popstate', on);
  }, []);
  return path;
}

/** Route in-app anchor clicks (including inside rendered Markdown) without a reload. */
export function interceptLinks(e: MouseEvent<HTMLElement>): void {
  const a = (e.target as HTMLElement).closest('a');
  const href = a?.getAttribute('href');
  if (!a || !href || !href.startsWith('/') || e.metaKey || e.ctrlKey || a.target === '_blank') return;
  e.preventDefault();
  navigate(href);
}
```

```tsx
function Tree({ nodes, source, current }: { nodes: TreeNode[]; source: string; current: string | null }) {
  return (
    <ul className="ml-2 border-l border-border pl-2 text-sm">
      {nodes.map((n) => n.children ? (
        <li key={`d-${n.name}`}>
          <details open={current?.startsWith(`${n.name}/`) ?? false}>
            <summary className="cursor-pointer py-0.5 text-muted">{n.name}/</summary>
            <Tree nodes={n.children} source={source} current={current?.split('/').slice(1).join('/') ?? null} />
          </details>
        </li>
      ) : (
        <li key={n.slug}>
          <a href={`/pages/${source}/${n.slug}`} className={`block truncate py-0.5 ${current === n.name ? 'font-semibold' : ''}`}>{n.name}</a>
        </li>
      ))}
    </ul>
  );
}

export function Pages({ me: _me, path }: { me: Me; path: string }) {
  const m = /^\/pages(?:\/([^/]+)(?:\/(.+))?)?$/.exec(path);
  const [sources, setSources] = useState<Source[] | null>(null);
  const source = m?.[1] ?? sources?.[0]?.id ?? null;
  const slug = m?.[2] ? decodeURIComponent(m[2]) : null;
  const [rows, setRows] = useState<PageRow[] | null>(null);
  const [q, setQ] = useState(''); const [hits, setHits] = useState<Hit[] | null>(null);
  const [page, setPage] = useState<Page | null>(null); const [error, setError] = useState<string | null>(null);

  useEffect(() => { api<{ sources: Source[] }>('/api/sources').then((r) => setSources(r.sources)); }, []);
  useEffect(() => {
    if (!source) return;
    setRows(null);
    api<{ pages: PageRow[] }>(`/api/pages?source=${source}`).then((r) => setRows(r.pages), (e) => setError(String(e.message)));
  }, [source]);

    if (!source || !slug) { setPage(null); return; }

    return () => clearTimeout(t);
  }, [q, source]);

  const tree = useMemo(() => buildTree((rows ?? []).map((r) => r.slug)), [rows]);
  if (sources === null) return <Spinner />;
  if (sources.length === 0) return <Empty>No sources you can read yet.</Empty>;

        {error && <Notice tone="danger">{error}</Notice>}
      </div>
      <div className="min-w-0 flex-1">

            <PageTitle>{String(page.frontmatter.title ?? page.slug)}</PageTitle>
            <div className="mb-4 flex flex-wrap gap-2 text-sm text-muted">
              <code>{page.slug}</code>

              </Card>
            )}
          </>
        ) : <Empty>Pick a page on the left, or search.</Empty>}
```

```tsx
import { useCallback, useEffect, useState } from 'react';
import { api, ApiError, type Me, type Role, type Source } from '../api.ts';
import { Button, Card, ConfirmButton, Empty, Field, Input, Notice, PageTitle, Select, Spinner, Table, TableWrap } from '../ui/index.tsx';

type Person = { id: string; email: string; name: string | null; role: Role; status: string; grants: { sourceId: string; agentReads: boolean }[] };
type PeopleData = { mode: 'personal' | 'company'; users: Person[]; sources: (Source & { ownerUserId: string | null })[] };

const message = (e: unknown) => {
  const code = e instanceof ApiError ? e.code : 'error';
  return ({ not_sole_member: 'Switch back only when you are the only person left.', company_mode_required: 'Switch to company mode first.',
    forbidden: 'You do not have permission for that.', user_exists: 'That person already has an account.' } as Record<string, string>)[code] ?? code;
};

export function People({ me, onChange }: { me: Me; onChange: () => void }) {
  const [d, setD] = useState<PeopleData | null>(null); const [err, setErr] = useState<string | null>(null);
  const [email, setEmail] = useState(''); const [role, setRole] = useState<Role>('member'); const [claimUrl, setClaimUrl] = useState<string | null>(null);
  const [newSource, setNewSource] = useState('');
  const load = useCallback(() => api<PeopleData>('/api/people').then(setD), []);
  useEffect(() => { load(); }, [load]);
  const act = async (f: () => Promise<unknown>) => { setErr(null); try { await f(); await load(); onChange(); } catch (e) { setErr(message(e)); } };

  if (!d) return <Spinner />;
  const owner = me.user.role === 'owner';
  const company = d.sources.filter((s) => s.kind === 'company' && s.status === 'active');
  const people = d.users.filter((u) => u.status !== 'disabled');

  return (
    <div className="flex flex-col gap-4">
      <PageTitle>People & access</PageTitle>
      {err && <Notice tone="danger">{err}</Notice>}
      <Card title={d.mode === 'company' ? 'Company mode' : 'Personal mode'}
        subtitle={d.mode === 'company' ? 'People can be invited. Each has a private space; company sources are shared by grant.' : 'This brain is yours alone. Switch to company mode to invite people.'}>
        {owner && <Button onClick={() => act(() => api('/api/people/mode', { json: { mode: d.mode === 'company' ? 'personal' : 'company' } }))}>
          Switch to {d.mode === 'company' ? 'personal' : 'company'} mode</Button>}
      </Card>
      {d.mode === 'company' && (<>
        <Card title="Invite someone">
          <form className="flex flex-wrap items-end gap-2" onSubmit={(e) => { e.preventDefault(); act(async () => {
            const r = await api<{ claimUrl: string }>('/api/people/invite', { json: { email, role } }); setClaimUrl(r.claimUrl); setEmail(''); }); }}>
            <Field label="Email"><Input type="email" value={email} onChange={(e) => setEmail(e.target.value)} required /></Field>
            <Field label="Role"><Select value={role} onChange={(e) => setRole(e.target.value as Role)}>
              <option value="member">member</option><option value="viewer">viewer</option>{owner && <option value="admin">admin</option>}</Select></Field>
            <Button variant="primary" type="submit">Create invite</Button>
          </form>
          {claimUrl && <Notice tone="ok" layout="stack">Send this link to them (valid 24 hours, works once):<code className="break-all">{claimUrl}</code>
            <Button onClick={() => navigator.clipboard.writeText(claimUrl)}>Copy link</Button></Notice>}
        </Card>
        <Card title="People and grants" subtitle="A tick lets that person read the company source. 'agent' means their agent reads it too.">
          <TableWrap><Table><thead><tr><th>Person</th><th>Role</th><th>Status</th>{company.map((s) => <th key={s.id}>{s.name}</th>)}<th /></tr></thead>
            <tbody>{people.map((u) => {
              const fixed = u.role === 'owner' || u.role === 'admin';
              return (<tr key={u.id}>
                <td>{u.name ?? u.email}<div className="text-muted text-sm">{u.email}</div></td>
                <td>{u.role === 'owner' || u.id === me.user.id ? u.role : (
                  <Select value={u.role} onChange={(e) => act(() => api(`/api/people/${u.id}/role`, { json: { role: e.target.value } }))}>
                    <option value="member">member</option><option value="viewer">viewer</option>{owner && <option value="admin">admin</option>}</Select>)}</td>
                <td>{u.status}</td>
                {company.map((s) => {
                  const g = u.grants.find((x) => x.sourceId === s.id);
                  return <td key={s.id}>{fixed ? 'all' : (
                    <label className="flex items-center gap-1 text-sm">
                      <input type="checkbox" checked={!!g} onChange={() => act(() => g
                        ? api(`/api/people/${u.id}/grants/${s.id}`, { method: 'DELETE' })
                        : api(`/api/people/${u.id}/grants/${s.id}`, { method: 'PUT', json: { agentReads: true } }))} />
                      {g && <span className="text-muted">{g.agentReads ? 'agent' : 'ui only'}</span>}
                    </label>)}</td>;
                })}
                <td>{u.role !== 'owner' && u.id !== me.user.id && (
                  <ConfirmButton label="Offboard" confirmLabel="Revoke all access?" onConfirm={() => act(() => api(`/api/people/${u.id}/offboard`, { json: {} }))} />)}</td>
              </tr>);
            })}</tbody></Table></TableWrap>
        </Card>
        <Card title="Company sources">
          {company.length === 0 ? <Empty>None.</Empty> : (
            <ul className="text-sm">{company.map((s) => (
              <li key={s.id} className="flex items-center gap-2 py-1">{s.name} <code className="text-muted">{s.id}</code>
                <ConfirmButton label="Archive" confirmLabel="Archive for everyone?" onConfirm={() => act(() => api(`/api/sources/${s.id}/archive`, { json: {} }))} /></li>))}</ul>)}
          <form className="mt-3 flex items-end gap-2" onSubmit={(e) => { e.preventDefault(); act(async () => { await api('/api/sources/company', { json: { name: newSource } }); setNewSource(''); }); }}>
            <Field label="New company source"><Input value={newSource} onChange={(e) => setNewSource(e.target.value)} required /></Field>
            <Button type="submit">Add</Button>
          </form>
        </Card>
      </>)}
    </div>
  );
}
```

```tsx
import { useEffect, useState } from 'react';
import { api, type Me } from '../api.ts';
import { ago } from '../lib.ts';
import { Button, Card, Empty, Notice, PageTitle, Table, TableWrap } from '../ui/index.tsx';

type Client = { purpose: string; writeSource: string | null; expiresAt: string | null };

export function Connect({ me, onChange }: { me: Me; onChange: () => void }) {
  const [say, setSay] = useState<string | null>(null);
  const [clients, setClients] = useState<Client[] | null>(null);
  useEffect(() => { api<{ clients: Client[] }>('/api/clients/me').then((r) => setClients(r.clients)); }, []);
  const agent = clients?.find((c) => c.purpose === 'agent');
  const soon = agent?.expiresAt && new Date(agent.expiresAt).getTime() - Date.now() < 14 * 86400_000;

  return (
    <div className="flex flex-col gap-4">
      <PageTitle>Connect your agent</PageTitle>
      <Card title="Set up an agent on a machine" subtitle="Creates a one-time key and the sentence to paste to your coding agent. Any previous unused key stops working.">
        <Button variant="primary" onClick={async () => setSay((await api<{ say: string }>('/api/me/join-key', { method: 'POST' })).say)}>Create join instructions</Button>
        {say && <div className="mt-3"><pre className="whitespace-pre-wrap break-all rounded bg-bg p-3 text-sm">{say}</pre>
          <Button onClick={() => navigator.clipboard.writeText(say)}>Copy</Button></div>}
      </Card>
      <Card title="Your agent's access">
        {clients === null ? null : !agent ? <Empty>No agent connected yet.</Empty> : (
          <>
            {soon && <Notice tone="warning">The agent token expires {ago(agent.expiresAt)}. Collectors rotate it automatically; if they are not running, ask your agent to run <code>mobrain client issue agent --register</code>.</Notice>}
            <TableWrap><Table><thead><tr><th>Client</th><th>Writes to</th><th>Expires</th></tr></thead>
              <tbody>{clients.filter((c) => c.purpose !== 'ui').map((c, i) => <tr key={i}><td>{c.purpose}</td><td>{c.writeSource ?? '-'}</td><td>{ago(c.expiresAt)}</td></tr>)}</tbody>
            </Table></TableWrap>
          </>
        )}
      </Card>
      {me.grants.length > 0 && (
        <Card title="Company sources your agent reads" subtitle="You can read these in this UI either way. Untick to keep one out of your agent's context.">
          <ul className="text-sm">{me.grants.map((g) => (
            <li key={g.sourceId} className="py-1"><label className="flex items-center gap-2">
              <input type="checkbox" checked={g.agentReads} onChange={async () => {
                await api(`/api/me/grants/${g.sourceId}`, { method: 'PATCH', json: { agentReads: !g.agentReads } }); onChange(); }} />
              {g.name}</label></li>))}</ul>
        </Card>
      )}
    </div>
  );
}
```
